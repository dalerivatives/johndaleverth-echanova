# Search engines: index the portfolio, stay out of the editor and the API.
User-agent: *
Allow: /

# The health endpoints are carved out BEFORE the /api/ block below, and the
# order matters. Plenty of uptime monitors fetch robots.txt and honour it, so
# a blanket "Disallow: /api/" quietly stops them pinging — the keep-alive
# never fires, the service sleeps, and the monitor reports nothing wrong
# because it never made a request in the first place. These three paths
# expose nothing but a status line.
Allow: /api/health
Allow: /api/health/db
Allow: /api/health/awake

# The icons, explicitly. Google will not show a favicon in search results
# for a URL it is not allowed to fetch, and a blanket Disallow one line
# further down is exactly how a site ends up with the grey globe.
Allow: /brand-icon.png
Allow: /brand-icon.ico
Allow: /brand-icon-512.png
Allow: /brand-icon-touch.png
Allow: /favicon.ico
Allow: /icon-192.png
Allow: /icon-512.png
Allow: /site.webmanifest

Disallow: /editor.html
Disallow: /api/

# Replace example.com with your real deployed address so search engines can
# find the sitemap (it must be an absolute URL).
Sitemap: https://example.com/sitemap.xml
